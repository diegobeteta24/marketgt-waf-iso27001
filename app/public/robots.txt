# MarketGT - robots.txt
# Universidad Mariano Galvez de Guatemala - Seguridad y Auditoria de Sistemas
#
# Este archivo NO es un control de seguridad: es una peticion que solo cumplen los
# rastreadores honestos. Lo que de verdad impide la indexacion es la cabecera
# X-Robots-Tag que emite App\Http\Middleware\DetectarCloaking, porque esa manda
# sobre cualquier <meta name="robots"> que un atacante logre inyectar en el HTML.
# Aqui se declara la intencion; alli se hace cumplir.
#
# La integridad de este archivo la vigila "php artisan seo:vigilar" cada cinco
# minutos contra la linea base sellada. Un "Disallow: /" colado bajo User-agent: *
# desindexaria la tienda completa en dias, y sin vigilancia nadie se enteraria
# hasta ver caer las ventas.
# =============================================================================

User-agent: *
Allow: /

# --- Resultados de busqueda interna -----------------------------------------
# Via clasica del envenenamiento por indexacion: el atacante busca su spam, la
# pagina de resultados lo refleja, el rastreador la indexa y el dominio de
# MarketGT acaba apareciendo en Google vendiendo lo que no vende.
Disallow: /buscar
Disallow: /busqueda
Disallow: /*?buscar=
Disallow: /*&buscar=
Disallow: /*?busqueda=
Disallow: /*?q=
Disallow: /*&q=
Disallow: /*?search=
Disallow: /*?termino=

# --- Paginas de sesion, compra y comprobantes -------------------------------

Disallow: /carrito

Disallow: /tienda/pago

Disallow: /pago
Disallow: /comprobante/
Disallow: /cuenta
Disallow: /pedidos
